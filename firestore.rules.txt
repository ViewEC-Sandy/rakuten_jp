rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function profileExists() {
      return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }
    function role() {
      return profileExists()
        ? get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role
        : "viewer";
    }
    function canEdit() { return signedIn() && (role() == "admin" || role() == "manager"); }
    function isAdmin() { return signedIn() && role() == "admin"; }

    match /users/{uid} {
      allow create: if signedIn() && request.auth.uid == uid && request.resource.data.role == "viewer";
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow update: if signedIn() && request.auth.uid == uid && request.resource.data.role == resource.data.role;
      allow update, delete: if isAdmin();
    }

    match /products/{id} {
      allow read: if signedIn();
      allow create, update: if canEdit();
      allow delete: if isAdmin();
    }
    match /sales/{id} {
      allow read: if signedIn();
      allow create, update: if canEdit();
      allow delete: if isAdmin();
    }
    match /platforms/{id} {
      allow read: if signedIn();
      allow create, update: if canEdit();
      allow delete: if isAdmin();
    }
    match /imports/{id} {
      allow read: if signedIn();
      allow create: if canEdit();
      allow update, delete: if isAdmin();
    }
  }
}
